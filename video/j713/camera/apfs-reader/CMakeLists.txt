cmake_minimum_required(VERSION 3.16)
project(m4_apfs_reader LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "Pinned upstream compatibility")
set(APFS_SOURCE_DIR "" CACHE PATH "Pinned apfs-fuse source with apfslib-readonly.patch applied")
if(NOT EXISTS "${APFS_SOURCE_DIR}/ApfsLib/ReaderLimits.h")
  message(FATAL_ERROR "Set APFS_SOURCE_DIR to the patched pinned source")
endif()
option(M4_APFS_TEST_IMAGES "Allow regular-file fixtures, never for installation" OFF)
option(M4_APFS_SANITIZERS "Enable ASan and UBSan for offline tests" OFF)
add_compile_options(-ffunction-sections -fdata-sections)
add_link_options(-Wl,--gc-sections)
if(M4_APFS_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
# Only the helper and static libraries are built. Upstream's FUSE, dump and
# apfsutil targets are excluded and require no FUSE headers or libraries.
add_subdirectory("${APFS_SOURCE_DIR}" upstream EXCLUDE_FROM_ALL)
add_executable(m4-apfs-reader helper.cpp)
target_include_directories(m4-apfs-reader PRIVATE "${APFS_SOURCE_DIR}")
target_compile_definitions(m4-apfs-reader PRIVATE _FILE_OFFSET_BITS=64)
target_compile_options(m4-apfs-reader PRIVATE -Wall -Wextra)
target_link_libraries(m4-apfs-reader PRIVATE apfs)
if(M4_APFS_TEST_IMAGES)
  target_compile_definitions(m4-apfs-reader PRIVATE M4_APFS_TEST_IMAGES=1)
  add_executable(fixture-layout EXCLUDE_FROM_ALL tests/fixture-layout.cpp)
  target_include_directories(fixture-layout PRIVATE "${APFS_SOURCE_DIR}")
endif()
if(M4_APFS_SANITIZERS)
  # ASan needs its shadow mapping and fault handlers. Only diagnostic builds
  # omit the production address-space limit and custom signal handler.
  target_compile_definitions(m4-apfs-reader PRIVATE M4_APFS_SANITIZER_BUILD=1)
endif()
